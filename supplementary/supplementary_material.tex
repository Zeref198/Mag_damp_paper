\subsection{Electric potential of B.C 4:}
\label{sec:bc4_approach}

From Eq.\,\meqref{eq:laplacian_psi} in the main article, the Laplace equation for the induced electric potential can be expanded to
\begin{gather}
	\pdv[2]{{\Psi}}{x} + \pdv[2]{{\Psi}}{y}
	+ \pdv[2]{{\Psi}}{z}= 0 .
\end{gather}
We write $\Psi(m,n) = A(x) \: B(y) \: C(z)$, where $A$, $B$ and $C$ are functions with respect to $x$, $y$ and $z$ accordingly
\begin{equation}
	\dfrac{1}{A} \dpdv[2]{A}{x}  + \dfrac{1}{B} \dpdv[2]{B}{y} 
	+ \dfrac{1}{C} \dpdv[2]{C}{z} = 0.
	\label{eq:laplace_ABC}
\end{equation}
Introducing constants $\xi_x$, $\xi_y$ and $\xi_z$ such that
\begin{equation}
	\xi_x^2 + \xi_y^2 + (-\xi_z^2) = 0 .
\end{equation}
At first glance from the given boundary conditions for B.C 4 in Table\,\ref{tab:table_BC_cases}, there is no straightforward way to solve the above Laplace equation using separation of variables, mainly due to the presence of non-zero boundary conditions in the $x$- and $y$-normal walls. This causes two of the constants to show exponential behavior in the solution, instead of the traditional system made out of one exponential and two oscillatory functions. Because the total sum of the balance must be zero, two of the constants need to have the opposite sign of the third. As a solution to this problem, the problem is to be divided into two constituent parts:
\begin{subequations}
	\begin{align}
	\RN{1} : \xi_x^2 + (-\xi_y^2) + (-\xi_z^2) = 0,
	\label{eq:laplace_m} 
	\\
	\RN{2} : (-\xi_x^2) + \xi_y^2 + (-\xi_z^2) = 0,
	\label{eq:laplace_n} 
	\end{align}
\end{subequations}
with Eqs.\,(\ref{eq:laplace_m} and \ref{eq:laplace_n}) solve the Laplace equation \eqref{eq:laplace_ABC} in a system with the following boundary conditions:
\begin{subequations}
	\begin{align}
		\RN{1} &: \partial_x \Psi\bigl|_{x=\pm 0.5L_x} = \partial_y\phi B_z,\;
		\partial_y \Psi\bigl|_{y=\pm 0.5L_y}= 0, \;
		\partial_z\Psi\bigl|_{z=-h}=0
		\label{eq:bc_m} 
		\\
		\RN{2} &: \partial_x \widehat\Psi\bigl|_{x=\pm 0.5L_x} = 0, \;
		\partial_y \Psi\bigl|_{y=\pm 0.5L_y}= -\partial_x\phi B_z,\;
		\partial_z\Psi\bigl|_{z=-h}=0
		\label{eq:bc_n} 
	\end{align}
\end{subequations}
We first solve for system \RN{1} considering the boundary conditions at $x=-0.5L_x$ and $x=0.5L_x$:
\begin{equation}
\pdv{\Psi}{x}\Bigl|_{x=-0.5L_x,0.5L_x} = \pdv{A}{x}\bigl(-0.5L_x,0.5L_x\bigr)B(y)C(z) = \pdv{\phi}{y}B_z\Bigl|_{x=-0.5L_x,0.5L_x}.
\end{equation}
The term $A(x)$ can be written as a sum of hyperbolic functions
\begin{equation}
	A(x) = A_{1x}\cosh{[\xi_x x]} + A_{2x}\sinh{[\xi_x x]},
\end{equation}
and at $x=-0.5L_x$:
\begin{align}
	\pdv{A}{y}\biggl|_{x=-0.5L_x} &= A_{1x}\xi_x\sinh{[-\xi_xL_x/2]}
	+ A_{2x}\xi_x\cosh{[-\xi_xL_x/2]},
	\\
	& = -\frac{n\pi}{L_y}\sin{\left[\frac{n\pi}{L_y}(y+0.5L_y)\right]} \cos{[0]},
\intertext{whereas at $x=0.5L_x$:}
	\pdv{A}{y}\biggl|_{x=0.5L_x} &= A_{1x}\xi_x\sinh{[\xi_xL_x/2]}
	+ A_{2x}\xi_x\cosh{[\xi_xL_x/2]},
	\\
	& = -\frac{n\pi}{L_y}\sin{\left[\frac{n\pi}{L_y}(y+0.5L_y)\right]} \cos{[m\pi]}.
\end{align}
When $m$ is odd,
\begin{equation}
2A_{2x}\xi_x\cosh{[\xi_xL_x/2]} = 0, \quad \therefore A_{2x} = 0 \Longrightarrow A(x) = A_{1x} \cosh{[\xi_x x]},
\end{equation}
and when $m$ is even,
\begin{equation}
-2A_{1x}\xi_x\sinh{[\xi_xL_x/2]} = 0, \quad \therefore A_{1x}=0 \Longrightarrow A(x) = A_{2x} \sinh{[\xi_x x]}.
\end{equation}
Therefore,
\begin{equation}
	A(x) = 
	A_{1x} \frac{1-(-1)^m}{2\sech{[\xi_x x]}} + A_{2x} \frac{1+(-1)^m}{2\csch{[\xi_x x]}}.
\end{equation}
At the $y$-normal walls,
\begin{equation}
	\pdv{B}{x} \: (-0.5L_y,0.5L_y) = 0,
\end{equation}
and therefore,
\begin{equation}
	B(y) = \sum_{a=0}^{\infty}
	B_{y} \cos{\biggl[ \frac{a\pi}{L_y}\Bigl(y + \frac{L_y}{2} \Bigr)\biggr]} .
\end{equation}
The boundary conditions at the bottom wall $(z=-h)$, are
\begin{equation}
	\pdv{C}{z}\bigl(z=-h\bigr) = 0,
\end{equation}
and at the free-surface $(z\approx0)$ is
\begin{equation}
	\pdv{C}{z}\bigl(z\approx 0) = 0.
\end{equation}
This leads to,
\begin{equation}
	C(z) = \sum_{b=0}^{\infty}
	 C_{z} \cos{\biggl[ \frac{b\pi}{h}\bigl(z + h\bigr)\biggr]}.
\end{equation}
Combining the coefficients such that
\begin{equation}
	\left(A_{1x} \frac{1-(-1)^m}{2} + A_{2x} \frac{1+(-1)^m}{2}\right)
	B_y\: C_z = \ddot{\mathcal{A}}_{ab},
\end{equation}
the solution for system \RN{1} is
\begin{equation}
	\Longrightarrow
	\sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	\ddot{\mathcal{A}}_{ab} 
	\left(\frac{1-(-1)^m}{2\sech{[\xi_x x]}} + \frac{1+(-1)^m}{2\csch{[\xi_x x]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_y}\biggl(y + \frac{L_y}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z+h\bigr)\biggr]} .
	\label{eq:sol_system1}
\end{equation}
Similarly, the solution for system \RN{2} is
\begin{equation}
	\Longrightarrow
	\sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	\ddot{\mathcal{B}}_{ab}
	\left(\frac{1-(-1)^n}{2\sech{[\xi_y y]}} + \frac{1+(-1)^n}{2\csch{[\xi_y y]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z+h\bigr)\biggr]} ,
\end{equation}
where
\begin{equation}
	\xi_x = \pi\sqrt{
	\biggl(\frac{a}{L_y}\biggr)^2 + \biggl(\frac{b}{h}\biggr)^2
	}, \quad
	\xi_y = \pi\sqrt{
		\biggl(\frac{a}{L_x}\biggr)^2 + \biggl(\frac{b}{h}\biggr)^2
	}.
\end{equation}
Combining the solutions of system \RN{1} and \RN{2}, we get
\begin{align}
	\nn
	\Psi &= 
	\sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	\ddot{\mathcal{A}}_{ab} 
	\left(\frac{1-(-1)^m}{2\sech{[\xi_x x]}} + \frac{1+(-1)^m}{2\csch{[\xi_x x]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_y}\biggl(y + \frac{L_y}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z+h\bigr)\biggr]}
	 \\
	 &+ 
	 \sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	 \ddot{\mathcal{B}}_{ab} 
	 \left(\frac{1-(-1)^n}{2\sech{[\xi_y y]}} + \frac{1+(-1)^n}{2\csch{[\xi_y y]}}
	 \right)
	 \cos{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	 \cos{\biggl[ \frac{b\pi}{h}\bigl(z+h\bigr)\biggr]} .
	 \label{eq:potential_initial}
\end{align}
We now need to determine the coefficients $\ddot{\mathcal{A}}_{ab}$ and $\ddot{\mathcal{B}}_{ab}$. This is achieved by projecting the boundary conditions in Table\,\ref{tab:table_BC_cases} (B.C 4) onto the oscillatory functions of Eq.\,\eqref{eq:potential_initial} using the orthogonality relation
\begin{equation}
	\int_{-0.5L}^{0.5L}
	\cos{\biggl[ \frac{a\pi}{L}\biggl(x + \frac{L}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{a'\pi}{L}\biggl(x + \frac{L}{2}\biggr)\biggr]}
	=
	\left\{
	\begin{array}{cc}
		L, & a=0,
		\\
		0.5L, & a\neq 0
	\end{array}
	\right. ,
\end{equation}
which yields a non-zero solution only for the condition $a=a'$ and thus, removing the double summations from the coefficients. This final expansions as given in Eqs.\,(\mref{eq:bc4_Acoeff} and \mref{eq:bc4_Bcoeff}) in the main article. The double sum in Eq.\,\eqref{eq:potential_initial} consists of real elements for all elements of the series except when $a=0$ and $b=0$ which raises an undefined value, both in the expression, and in the coefficients $\ddot{\mathcal{A}}_{ab}$ and $\ddot{\mathcal{B}}_{ab}$. This gives rise to constant terms unique to the case B.C 4, which arise from the boundary conditions in the $x$- and $y$-normal walls
\begin{equation}
	\ddot{\mathcal{A}}_{0} = \int \partial_y\phi_y B_z, \quad \& \quad
	\ddot{\mathcal{B}}_{0} = \int -\partial_x \phi_x B_z .
\end{equation}
The expressions of $\ddot{\mathcal{A}}_{0}$ and $\ddot{\mathcal{B}}_{0}$ are elaborated in Eq.\,\meqref{eq:bc4_constants}. This results in the final expression of the induced electric potential
\begin{align}
	\nn
	\Psi (\ii{B.C 4})&= 
	\ddot{\mathcal{A}}_{0}x - \frac{\ddot{\mathcal{A}}_{0} \bigl(-1^n + (-1)^{m+n+1} \bigr) }{2L_x}
	\left(x- \frac{L_x}{2}\right)^2
	+
	\ddot{\mathcal{B}}_{0}y - \frac{\ddot{\mathcal{B}}_{0} \bigl(-1^m + (-1)^{m+n+1} \bigr) }{2L_y}
	\left(y- \frac{L_y}{2}\right)^2
	\\
	\nn
	&+
	\sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	\ddot{\mathcal{A}}_{ab} 
	\left(
	\frac{1-(-1)^m}{2\sech{[\xi_x x]}} + \frac{1+(-1)^m}{2\csch{[\xi_x x]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_y}\biggl(y + \frac{L_y}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z +  h\bigr)\biggr]} 
	\\
	&+
	\sum_{a=0}^{\infty} \sum_{b=0}^{\infty}
	\ddot{\mathcal{B}}_{ab} 
	\left(
	\frac{1-(-1)^n}{2\sech{[\xi_y y]}} + \frac{1+(-1)^n}{2\csch{[\xi_y y]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z +  h\bigr)\biggr]}
.
\end{align}

\subsection{Electric potential of B.C 2:}

In case B.C 2, all the boundary conditions are the same as in B.C 4 except for the $x$-normal walls. This results in the solution of system \RN{1} in Eq.\,\eqref{eq:sol_system1} being equal to zero. The resultant expression for the electric potential is therefore:
%%%%%%%%%%%%%
\begin{gather}
	\Psi (\ii{B.C 2})= 
	\sum_{a,b=0}^{\infty}
	\grave{\mathcal{B}}_{ab} 
	\left(
	\frac{1-(-1)^n}{2\sech{[\grave\xi_y y]}} + \frac{1+(-1)^n}{2\csch{[\grave\xi_y y]}}
	\right)
	\sin{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	\cos{\biggl[ \frac{b\pi}{h}\bigl(z + h\bigr)\biggr]}
	,
\end{gather}
where $\grave{\mathcal{B}}_{ab}$ is obtained using the orthogonality condition, which yields the expression presented in Eq.\,\eqref{eq:bc2_B} in the main article.

\subsection{Electric potential of B.C 3:}

Case B.C 3 is also a modification of B.C 4, with the exception of the conducting bottom wall, which creates an asymmetric boundary condition along the $z$-axis. This is due to the free-surface acting as an insulating wall and the bottom wall being a perfect conductor,
%%%%%%%%%%%%
\begin{equation}
	\pdv{C}{z}\bigl(z\approx 0\bigr) = 0, \quad C(z=-h) = 0,
\end{equation}
which results in 
\begin{equation}
		C(z) = \sum_{b=0}^{\infty}
	C_{z} \sin{\biggl[(2b+1)p_h \bigl(z + h\bigr)\biggr]}, \quad \ii{with} \;\; p_h=\frac{\pi}{2h}.
\end{equation}
Using the approach outlined in Sec.\,\ref{sec:bc4_approach} for case B.C 4, we obtain the following expression for the electric potential:
\begin{align}
	\nn
	\Psi (\ii{B.C 3})&=
	\sum_{\substack{a, b \geq 0 \\ 
			(a,b)\neq (0,0) \\
			a\neq n}
	}^{\infty}
	\tilde{\mathcal{A}}_{ab}
	\left(
	\frac{1-(-1)^m}{2\sech{\bigl[\tilde\xi_x x \bigr]}} + \frac{1+(-1)^m}{2\csch{\bigl[\tilde\xi_x x \bigr]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_y}\biggl(y + \frac{L_y}{2}\biggr)\biggr]}
	\sin{\biggl[ (2b+1)p_h\bigl(z + h\bigr)\biggr]} 
	\\
	&+
	\sum_{\substack{a, b \geq 0 \\ (a,b) 
			\neq (0,0)\\
			a\neq m		
	}}^{\infty}
	\tilde{\mathcal B}_{ab}
	\left(
	\frac{1-(-1)^n}{2\sech{\bigl[\tilde\xi_y y\bigr]}} + \frac{1+(-1)^n}{2\csch{\bigl[\tilde\xi_y y\bigr]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	\sin{\biggl[ (2b+1)p_h\bigl(z + h\bigr)\biggr]}
	,
\end{align}
where the coefficients $\tilde{\mathcal{A}}_{ab}$ and $\tilde{\mathcal{B}}_{ab}$ are obtained from the orthogonality condition. This gives expressions presented in Eqs.\,(\ref{eq:bc3_A} and \ref{eq:bc3_B}) in the main article. Similar to B.C 4, we need to incorporate additional constants to satisfy the boundary conditions when the summation indices $a,b=0$. This results in the terms $\tilde{\mathcal{A}}_{0}$ and $\tilde{\mathcal{B}}_{0}$, which are presented in Eqs.\,(\ref{eq:bc3_constantsA0} and \ref{eq:bc3_constantsB0}) in the main article. The final expression of the electrical potential is given by
\begin{align}
	\nn
	\Psi (\ii{B.C 3})&= 
	\sin{[(z+h)p_h]} \left(
	\dfrac{ \tilde{\mathcal A_0} \bigl((-1)^{m} - (-1)^n \bigr)\:
		\sinh{[p_h x]}}{2p_h \cosh{[0.5p_hL_x]}} 
	- 
	\dfrac{
		\tilde{\mathcal A_0} \bigl((-1)^{m} + (-1)^n \bigr) \:
		\cosh{[p_h x]}}{2p_h \sinh{[0.5p_hL_x]}}
	\right.
	\\
	\nn
	&+
	\left.
	\dfrac{ \tilde{\mathcal B_0} \bigl((-1)^{n} - (-1)^m \bigr)\:
		\sinh{[p_h y]}}{2p_h \cosh{[0.5p_hL_y]}} 
	- 
	\dfrac{
		\tilde{\mathcal B_0} \bigl((-1)^{n} + (-1)^m \bigr) \:
		\cosh{[p_h y]}}{2p_h \sinh{[0.5p_hL_y]}}
	\right)
	\\
	\nn
	&+
	\sum_{\substack{a, b \geq 0 \\ 
			(a,b)\neq (0,0) \\
			a\neq n}
	}^{\infty}
	\tilde{\mathcal{A}}_{ab}
	\left(
	\frac{1-(-1)^m}{2\sech{\bigl[\tilde\xi_x x \bigr]}} + \frac{1+(-1)^m}{2\csch{\bigl[\tilde\xi_x x \bigr]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_y}\biggl(y + \frac{L_y}{2}\biggr)\biggr]}
	\sin{\biggl[ (2b+1)p_h\bigl(z + h\bigr)\biggr]} 
	\\
	&+
	\sum_{\substack{a, b \geq 0 \\ (a,b) 
			\neq (0,0)\\
			a\neq m		
	}}^{\infty}
	\tilde{\mathcal B}_{ab}
	\left(
	\frac{1-(-1)^n}{2\sech{\bigl[\tilde\xi_y y\bigr]}} + \frac{1+(-1)^n}{2\csch{\bigl[\tilde\xi_y y\bigr]}}
	\right)
	\cos{\biggl[ \frac{a\pi}{L_x}\biggl(x + \frac{L_x}{2}\biggr)\biggr]}
	\sin{\biggl[ (2b+1)p_h\bigl(z + h\bigr)\biggr]}
	.
\end{align}

\section{Supplementary material: The mechanical energy balance in the bulk} 
\label{supplB}

In this section, we provide the detailed methodology of deriving the Ohmic damping rate $\lambda_{\rm o}$ using the mechanical energy balance, written once again in its complete form here:
\begin{equation}
	\underbrace{ \dv{}{t}\left( \frac{1}{2} 
	\int_{V} \rho |\h u|^2 \mathrm dV  \right)
}_{\rm E_{kin}}
	+
	\underbrace{
	\dv{}{t}\left( \int_{\mathcal{S}} 
	\frac{1}{2} \rho g\eta^2  \: \mathrm dS  
	\right)
}_{\rm E_{pot}}
	= 
	\underbrace{
	\int_{V} \h u \cdot (\pmb{\mathcal{J}} \times B_z\h e_z)\: \mathrm dV
}_{\rm D_{bulk}} ,
	\label{eq:balance_ohmic_complete}
\end{equation}
where each integral term $\rm E_{kin}$ is the time rate of change kinetic energy of the bulk, $\rm E_{pot}$ is the time rate of change potential energy of the free-surface and $\rm D_{bulk}$, is the energy per unit volume transferred between the electromagnetic field and the moving fluid. On expanding $\pmb{\mathcal{J}}$ in the scalar triple product
\begin{align} 
	\nn
	\h u \cdot (\pmb{\mathcal{J}} \times B_z\h e_z) 
	&= -\pmb{\mathcal{J}} \cdot ( \h u \times B_z\h e_z) ,
	\\ 
	\nn
	&= - \left[
	-\sigma\bm\nabla\Psi(\h u\times B_z\h e_z) 
	+ \sigma\norm{\h u \times B_z\h e_z}^2
	\right],
	\\
	& = -\pmb{\mathcal{J}}\cdot \bm\nabla\Psi - \frac{\pmb{\mathcal{J}}^2}{\sigma} .
	\label{eq:ohmic_expansion_nabla}
\end{align}
Substituting Eq.\,\eqref{eq:ohmic_expansion_nabla} in \eqref{eq:balance_ohmic_complete}, the term $\pmb{\mathcal{J}}\cdot \bm\nabla\Psi$ on the right hand integral expands to
\begin{align}
\int_{V} \pmb{\mathcal{J}}\cdot \bm\nabla\Psi
	&=
	\int_{\partial V}  \Psi \pmb{\mathcal J} \cdot \h n \mathrm \:\mathrm dS
	- \int_{V}  \Psi \:
	{\bm\nabla \cdot \pmb{\mathcal J}} \quad\mathrm dV,
	\label{eq:integral_J_nablaPsi}
\intertext{where,}
	\int_{\partial V}  \Psi \pmb{\mathcal J} \cdot \h n \mathrm \:\mathrm dS
	&= 
	\int_{\Sigma} \Psi \pmb{\mathcal J} \cdot \h n \mathrm \:\mathrm dS
	+
	\int_{\mathcal{S}} 
	\Psi \pmb{\mathcal J} \cdot \h n \mathrm \:\mathrm dS.
	\label{eq:integral_Jdotn}
\end{align} 
In Eq.\,\eqref{eq:integral_J_nablaPsi}, $\bm\nabla \cdot \pmb{\mathcal J} = 0$, by virtue of conservation of charge. In the following Eq.\,\eqref{eq:integral_Jdotn}, the normal currents $\pmb{\mathcal J} \cdot \h n=0$ at insulating walls, whereas $\Psi=0$ at conducting walls. These transformations reduce the complete balance equation to the one presented in Eq.\,\meqref{eq:balance_ohmic}, in the main article. Therefore, $\rm D_{bulk}$ is expanded as follows:
\begin{align}
	\nn
	\ii D_{\rm bulk} &= \int_{V} 
	\bm\nabla\phi \cdot 
	\left( \sigma\left[ -\bm\nabla\Psi + \bm\nabla\phi \times B_z\h e_z \right]
	\times B_z\h e_z \right) \mathrm dV,
	\\
	\nn
	&=  \int_{V}  \bm\nabla\phi \cdot \bigl(
	-\sigma\bm\nabla\Psi \times B_z\h e_z \bigr) \mathrm dV
	- 
	 \int_{V} \norm{\bm\nabla\phi \times B_z\h e_z}^2 \: \mathrm dV,
	\\
	& = \mathcal{Q}_{\rm corr} - \mathcal{Q}_{\rm ind}. 
	\qquad \{ \ii{see Eqs.\,(\mref{eq:qind} and \mref{eq:Qcorr_integral})} \}
	\label{eq:Dbulk_result}
\end{align}
The total dissipation averaged over a single oscillation cycle is
\begin{equation}
	\ii D_{\rm bulk} = \frac{1}{2} \bigl(\mathcal{Q}_{\rm corr} - \mathcal{Q}_{\rm ind} \bigr).
	\label{eq:Dbulk_avg}
\end{equation}
\\
\indent The flow velocity $\h u$ and free-surface elevation $\eta$ decay proportional to $\ii e^{-\lambda_{\rm o}t}$ with respect to time $t$. For a standing wave, the total energy considering wave decay is the sum of the kinetic and potential energy with a phase difference, given by:
\begin{equation}
	\ii E_{\rm kin} + \ii E_{\rm pot} = 
	-\lambda_{\rm o} \frac{1}{2} \int_{V} \rho |\h u|^2 \mathrm dV
	\; \cos{[ \omega t ]}^2 
	-  \lambda_{\rm o} \frac{1}{2} \int_{\mathcal{S}}
	\rho g\eta^2 \mathrm dS
	\; \cos{[ \omega t  + 0.5\pi]}^2 .
	\label{eq:kin+pot_integrals}
\end{equation}
In the above equation, the integral 
\begin{equation}
	\int_{\mathcal{S}}
	\rho g\eta^2 \mathrm dS =
	 p|_{z=0} \eta \: \mathrm dS,
	\label{eq:young_laplace_integral}
\end{equation}
which is hydrostatic pressure condition that is obtained from the kinematic-dynamic coupling of an inviscid, irrotational flow of a lightly deformed free-surface. On expanding $\eta$ using Eq.\,\meqref{eq:eta}, we obtain the expression for $K$ as per Eq.\,\meqref{eq:K_eq} in the article. Similarly, using the vector identity  $\bm\nabla\phi \cdot \bm\nabla\phi = \bm\nabla\cdot(\phi \bm\nabla\phi) - \phi\bm\nabla\cdot \bm\nabla\phi$, we write
\begin{align}
	\nn
	\int_{V} \rho |\bm\nabla \phi|^2 \mathrm dV &=
	\int_{V} \left(
	\bm\nabla \cdot \phi\bm\nabla\phi - \phi\nabla^2\phi \right) \mathrm dV,
	\\
	&= 
	\int_{\partial V} 
	\rho\phi\bm\nabla\phi\cdot \h n \:\mathrm dS.
	\label{eq:grad_phi_dotn}
\end{align}
The above integral vanishes at the rigid walls $\Sigma$ due to the impermeability condition $\h u \cdot \h n=0$. At the free-surface, $\bm\nabla\phi\cdot \h n|_{z=0} = \partial_z\phi$, and from Eqs.\,(\mref{eq:eta} and \mref{eq:vel_potential}) in the main article, the vertical component of the velocity potential is expressed as $\partial_z\phi = \ii i \omega \phi$. Therefore, Eq.\,\eqref{eq:grad_phi_dotn} is
 \begin{align}
 	\nn
 	\int_{V}\rho |\bm\nabla \phi|^2 \mathrm dV &= \int_{\mathcal{S}} \left(
 	\rho(\ii i\omega)\phi
 	\right)|_{z=0} \: \eta \:\mathrm dS,
 	\\
 	&= \int_{\mathcal{S}}
 	 p|_{z=0} \: \eta \: \mathrm dS = K, 
 	\qquad \{ \because p = \ii i\omega\phi \}.
 	\label{eq:rho_u2_K}
 \end{align}
From Eqs.\,(\ref{eq:Dbulk_avg}, \ref{eq:kin+pot_integrals}, \ref{eq:young_laplace_integral} and \ref{eq:rho_u2_K}), we obtain the following relation for $\lambda_{\rm o}$:
\begin{gather}
	-\lambda_{\rm o} K \cos{[\omega t]}^2 - \lambda_{\rm o} K \sin{[\omega t]}^2  =   \frac{1}{2} \bigl(\mathcal{Q}_{\rm corr} - \mathcal{Q}_{\rm ind} \bigr)
	\\[10pt]
	\therefore 
	\lambda_{\rm o} = \dfrac{\mathcal{Q}_{\rm ind} - \mathcal{Q}_{\rm corr}}{2K}
\end{gather}


\section{Supplementary material: The mechanical energy balance at the boundary} 
\label{supplC}

Like in Sec.\,\ref{supplB}, we apply the mechanical energy balance but for the dissipation at the boundary due to viscous and magnetic effects. We calculate the damping rate due to boundary effects $\lambda_{\rm b}$, which comprises of the damping rate due to the Hartmann layer at the top and bottom, and the Shercliff layer at the sidewalls. To summarize, the mechanical energy balance of a standing wave, considering dissipation at the boundaries is given by
\begin{equation}
	\underbrace{
	\dv{}{t}\left(  \frac{1}{2} 
	\int_{V} \rho |\h u|^2 \mathrm dV  \right)
}_{\ii E_{\rm kin}}
	+
	\underbrace{
	\dv{}{t}\left( \int_{\mathcal{S}} 
	\frac{1}{2} \rho g\eta^2  \: \mathrm d\mathcal{S}  
	\right)
}_{\ii E_{\rm pot}}
	= 
	\underbrace{
	- 2\rho\nu
	\int_{V} \bm\varepsilon \bm{\mathbin{:}} \bm\varepsilon  \: \mathrm dV
}_{\ii D_{\rm b}} ,
	\label{eq:balance_boundary_suppl}
\end{equation}
The terms $\ii E_{\rm kin}$ and $\ii E_{\rm pot}$ in Eq.\,\eqref{eq:balance_boundary_suppl} only consist of bulk quantities like $\h u$ without the corrections, which in this particular case, decay proportional to $\ii e^{-\lambda_{\rm b} t}$. The boundary damping rate consists of $\lambda_{\rm b} = \lambda_{\rm Ha}(\ii{wall}) + \lambda_{\rm Sh}(\ii{sidewall})$, each of which we calculate separately. Following the steps outlined in Sec.\,\ref{supplB}, the total energy during the cycle is
\begin{equation}
	\ii E_{\rm kin} + \ii E_{\rm pot} = -\lambda_* K,
	\label{eq:ekin_epot_k}
\end{equation}
with 
\begin{equation}
	K = 0.25\mathcal{A}^2_{mn}\epsilon_{mn}\rho g L_xL_y
	\;\ii e^{-2\lambda_*t} ,
	\quad 
	\lambda_* = \lambda_{\rm Ha}, \lambda_{\rm Sh}, \quad
	\epsilon_{r} =
	\biggl\{ \begin{array}{rc}
		2,  & r= 0 \\
		1,  & r\neq 0
	\end{array} .
\end{equation}
On the right hand side of Eq.\,\eqref{eq:balance_boundary_suppl}, expanding the strain rate tensor $\bm\varepsilon = 0.5(\bm\nabla\h u + \bm\nabla\h u^{\rm T})$ in the double dot product gives
\begin{equation}
	2 \: \bm\varepsilon \bm{\mathbin{:}} \bm\varepsilon = 
	\norm{\bm\nabla \times \h u}^2 +
	2\bm\nabla \cdot \left( (\h u \cdot \bm\nabla )\h u\right)
\end{equation} 
The dissipation term $\ii D_{\rm b}$ now takes the form of the Lamb's dissipation integral
\begin{align}
	\ii D_{\rm b} &= 
	- \rho\nu
	\int_{V} \norm{\bm\nabla \times \h u}^2  \: \mathrm dV
	-2  \rho\nu
	\int_{\partial V} 
	(\h u \cdot \bm\nabla )\h u \cdot \h n \:\mathrm dS,
\end{align}
which consists of the dissipation due to rotational stresses at the boundary and the higher order irrotational stresses in the bulk. We calculate only calculate the rotational stresses at the boundary, where only the corrections contribute. The boundary dissipation is therefore:
\begin{equation}
	\ii D_{\rm b} = - \rho\nu
	\int_{V} \norm{\bm\nabla \times \overbar{\h u}_{\perp}}^2  \: \mathrm dV,
	\label{eq:dissipation_rotational_stress}
\end{equation}
where tangential velocity corrections in the Hartmann nd Shercliff layers are
\begin{equation}
	{\overbar{\h u}}_{\perp} (\ii{Ha})=
	-\bm\nabla \phi 
	\ii e^{-
		\dfrac{\sqrt{1 + \sqrt{1 + \tau^2\omega^2}}}{\sqrt{2\tau\nu}}
		\zeta
	},
	\qquad
	{\overbar{\h u}}_{\perp} (\ii{Sh})=
	-\bm\nabla \phi
	\ii e^{-
		\dfrac{\sqrt{1 + \sqrt{1 + \tau h^2\omega^2/\nu}}}
		{\sqrt{2\sqrt{\tau\nu}}}
		\zeta
	}.
	\label{eq:boundary_velocities}
\end{equation}
Here, the wall normal coordinate $\zeta$ is the spatial coordinate normal to the wall that reduces the boundary layer corrections to zero away from the wall. It is given by
\begin{equation}
	\zeta = \begin{cases}
		z + h, \quad &\text{near } z=-h, \\
		\dfrac{L_x}{2} - x, \quad &\text{near } x = L_x/2, \qquad\qquad
		x + \dfrac{L_x}{2}, \quad \text{near } x = -L_x/2 \\[5pt]
		\dfrac{L_y}{2} - y, \quad &\text{near } y = L_y/2, \qquad\qquad
		y + \dfrac{L_y}{2}, \quad \text{near } y = -L_y/2,
	\end{cases} .
\end{equation}
Substituting the velocity corrections at the walls given in Eq.\,\eqref{eq:boundary_velocities} into Eq.\,\eqref{eq:dissipation_rotational_stress}, we get
\begin{equation}
	\ii D_{\rm Ha} = 
	-\frac{\rho\nu\omega^2}{4\sqrt{2}}
	\sqrt{
		\dfrac{1}{\tau\nu} + 
		\sqrt{ \dfrac{1}{\tau^2\nu^2} + \dfrac{\omega^2}{\nu^2} } 
	}  
	\Biggl(
	\frac{\epsilon_{mn} L_xL_y}{2 \: \text{sinh}^2[kh]}
	\Biggr) \ii{e}^{-2\lambda_{\rm Ha}\: t},
\end{equation}
which is the dissipation due to the Hartmann layer at the top and bottom wall, and
\begin{align}
	\nn
	\ii D_{\rm Sh}  = 
	-\frac{\rho\nu\omega^2}{\sqrt{2} \; k^2}
	\sqrt{ \frac{1}{h\sqrt{\tau\nu}} + 
		\sqrt{ \frac{1}{h^2\tau\nu} + \frac{\omega^2}{\nu^2}}}
	\Biggl(
	&\epsilon_n 
	\biggl[
	\frac{h(n^2\pi^2  - k^2L_y^2)}{4L_y\:\text{sinh}^2[kh]}
	+ \frac{(n^2\pi^2 + k^2L_y^2)}{4L_yk\:\text{tanh}[kh]}
	\biggr]
	\\
	 + \:
	&\epsilon_m
	\biggl[
	\frac{h(m^2\pi^2  - k^2L_x^2)}{4L_x\:\text{sinh}^2[kh]}
	+ \frac{(m^2\pi^2 + k^2L_x^2)}{4L_x \: k\:\text{tanh}[kh]}
	\biggr]
	\Biggr) \ii{e}^{-2\lambda_{\rm Sh} t} ,
	\label{eq:suppl_shercliff_dissip}
\end{align}
which is the dissipation due to the Shercliff layer at all four sidewalls. For most industrial application where $B_z < SI{3}{\tesla}$, the dissipation term from Eq.\,\eqref{eq:suppl_shercliff_dissip} simplifies to
\begin{align}
	\nn
	\ii D_{\rm Sh} 
	\approx 
	-\frac{\rho\nu\omega^2}{\sqrt{2} \; k^2}
	\sqrt{\frac{\omega}{\nu}}
	\Biggl(
	&\epsilon_n 
	\biggl[
	\frac{h(n^2\pi^2  - k^2L_y^2)}{4L_y\:\text{sinh}^2[kh]}
	+ \frac{(n^2\pi^2 + k^2L_y^2)}{4L_yk\:\text{tanh}[kh]}
	\biggr]
	\\
	+\:
	&\epsilon_m
	\biggl[
	\frac{h(m^2\pi^2  - k^2L_x^2)}{4L_x\:\text{sinh}^2[kh]}
	 + 
	\frac{(m^2\pi^2 + k^2L_x^2)}{4L_x \: k\:\text{tanh}[kh]}
	\biggr]
	\Biggr) \ii{e}^{-2\lambda_{\rm Sh} t},
\end{align}
which is the dissipation due rotational stresses solely due to the viscous boundary layer.
\\
\indent From Eqs.\,(\ref{eq:ekin_epot_k} and \ref{eq:dissipation_rotational_stress}), we get the expression for $\lambda_{\rm b}$, in its complete form as follows:
\begin{equation}
	\lambda_{\rm b} = \lambda_{\rm Ha} + \lambda_{\rm Sh} 
	=
	\frac{\ii D_{\rm Ha}
	+ \ii D_{\rm Sh} }{K} 
\end{equation}
